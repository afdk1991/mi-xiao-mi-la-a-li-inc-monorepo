# GEXM 跨平台网站预览功能实现方案

## 一、方案概述

本方案旨在构建一套完整的跨平台网站预览功能，确保 GEXM 平台在桌面端、移动端及平板端等不同设备的各种屏幕尺寸、分辨率和浏览器环境下均能正确显示和正常运行。方案包含响应式设计验证、跨浏览器兼容性测试以及不同设备类型的模拟预览功能。

| 项 | 值 |
|----|----|
| 方案版本 | v1.0 |
| 创建日期 | 2026-07-15 |
| 适用平台 | PC 管理后台、移动端 H5 应用 |
| 核心技术栈 | Playwright + Vue3 + Vite |

### 6.3 全量预览脚本 (preview-all.js)

```javascript
const { chromium, firefox, webkit } = require('playwright');
const { pcDevices, h5Devices, browsers } = require('./config/devices');
const { pcRoutes, h5Routes } = require('./config/routes');

const BROWSER_MAP = { chromium, firefox, webkit };

async function previewPlatform(platform, browserName) {
  const devices = platform === 'pc' ? pcDevices : h5Devices;
  const routes = platform === 'pc' ? pcRoutes : h5Routes;
  const baseUrl = platform === 'pc' ? 'http://localhost:5173' : 'http://localhost:5174';
  const browser = BROWSER_MAP[browserName];
  
  const launchOptions = browsers.find(b => b.name === browserName)?.launchOptions || {};
  const browserInstance = await browser.launch({ ...launchOptions, headless: true });
  
  const results = [];
  
  for (const device of devices) {
    for (const route of routes) {
      const context = await browserInstance.newContext({
        viewport: device.viewport,
        deviceScaleFactor: device.deviceScaleFactor,
        isMobile: device.isMobile,
        hasTouch: device.hasTouch,
        userAgent: device.userAgent,
      });
      
      const page = await context.newPage();
      const url = platform === 'h5' ? `${baseUrl}/#${route.path}` : `${baseUrl}${route.path}`;
      
      try {
        await page.goto(url, { waitUntil: 'networkidle' });
        
        const metrics = await page.evaluate(() => ({
          scrollWidth: document.documentElement.scrollWidth,
          clientWidth: document.documentElement.clientWidth,
          hasHorizontalOverflow: document.documentElement.scrollWidth > document.documentElement.clientWidth,
          loadTime: performance.timing.loadEventEnd - performance.timing.navigationStart,
        }));
        
        const screenshotPath = `preview/results/screenshots/${platform}-${browserName}-${device.name.replace(/\s/g, '-')}-${route.name}.png`;
        await page.screenshot({ path: screenshotPath, fullPage: true });
        
        results.push({
          platform,
          browser: browserName,
          device: device.name,
          viewport: `${device.viewport.width}×${device.viewport.height}`,
          route: route.name,
          url,
          status: metrics.hasHorizontalOverflow ? 'FAIL' : 'PASS',
          metrics,
          screenshot: screenshotPath,
        });
        
        console.log(`[${browserName}] ${platform.toUpperCase()} ${device.name} ${route.name}: ${metrics.hasHorizontalOverflow ? '❌ OVERFLOW' : '✅ OK'} (${metrics.loadTime}ms)`);
      } catch (error) {
        results.push({
          platform,
          browser: browserName,
          device: device.name,
          route: route.name,
          url,
          status: 'ERROR',
          error: error.message,
        });
        
        console.error(`[${browserName}] ${platform.toUpperCase()} ${device.name} ${route.name}: ❌ ERROR - ${error.message}`);
      }
      
      await context.close();
    }
  }
  
  await browserInstance.close();
  return results;
}

async function main() {
  const args = process.argv.slice(2);
  const platform = args.includes('--platform') ? args[args.indexOf('--platform') + 1] : 'all';
  
  console.log('='.repeat(60));
  console.log('GEXM Cross-Platform Preview Tool');
  console.log('='.repeat(60));
  console.log('');
  
  const allResults = [];
  
  if (platform === 'all' || platform === 'pc') {
    console.log('=== PC 管理后台预览 ===');
    for (const browser of browsers) {
      console.log(`\n[${browser.name.toUpperCase()}]`);
      const results = await previewPlatform('pc', browser.name);
      allResults.push(...results);
    }
  }
  
  if (platform === 'all' || platform === 'h5') {
    console.log('\n=== 移动端 H5 预览 ===');
    for (const browser of browsers) {
      console.log(`\n[${browser.name.toUpperCase()}]`);
      const results = await previewPlatform('h5', browser.name);
      allResults.push(...results);
    }
  }
  
  console.log('\n'.repeat(2));
  console.log('='.repeat(60));
  console.log('Preview Summary');
  console.log('='.repeat(60));
  
  const passed = allResults.filter(r => r.status === 'PASS').length;
  const failed = allResults.filter(r => r.status === 'FAIL').length;
  const errors = allResults.filter(r => r.status === 'ERROR').length;
  
  console.log(`Total: ${allResults.length}`);
  console.log(`Passed: ${passed}`);
  console.log(`Failed (overflow): ${failed}`);
  console.log(`Errors: ${errors}`);
  
  if (failed > 0) {
    console.log('\nFailed Tests:');
    allResults.filter(r => r.status === 'FAIL').forEach(r => {
      console.log(`  - ${r.platform.toUpperCase()} | ${r.browser} | ${r.device} | ${r.route}`);
    });
  }
  
  if (errors > 0) {
    console.log('\nErrors:');
    allResults.filter(r => r.status === 'ERROR').forEach(r => {
      console.log(`  - ${r.platform.toUpperCase()} | ${r.browser} | ${r.device} | ${r.route}: ${r.error}`);
    });
  }
  
  const fs = require('fs');
  const reportPath = `preview/results/reports/preview-report-${Date.now()}.json`;
  fs.writeFileSync(reportPath, JSON.stringify(allResults, null, 2));
  console.log(`\nReport saved to: ${reportPath}`);
}

main().catch(console.error);
```

## 七、视觉对比与报告生成

### 7.1 视觉对比脚本 (visual-compare.js)

```javascript
const { chromium } = require('playwright');
const fs = require('fs');
const path = require('path');

async function compareBaselines() {
  const baselineDir = 'preview/results/baselines';
  const screenshotDir = 'preview/results/screenshots';
  const results = [];
  
  if (!fs.existsSync(baselineDir)) {
    console.log('No baseline directory found. Creating initial baselines...');
    await createBaselines();
    return;
  }
  
  const browser = await chromium.launch({ headless: true });
  const context = await browser.newContext();
  
  const baselineFiles = fs.readdirSync(baselineDir).filter(f => f.endsWith('.png'));
  
  for (const baselineFile of baselineFiles) {
    const baselinePath = path.join(baselineDir, baselineFile);
    const screenshotPath = path.join(screenshotDir, baselineFile);
    
    if (!fs.existsSync(screenshotPath)) {
      results.push({
        file: baselineFile,
        status: 'MISSING',
        message: 'Screenshot not found',
      });
      continue;
    }
    
    const baselinePage = await context.newPage();
    await baselinePage.setViewportSize({ width: 1920, height: 1080 });
    await baselinePage.goto(`file:///${path.resolve(baselinePath)}`);
    
    const screenshotPage = await context.newPage();
    await screenshotPage.setViewportSize({ width: 1920, height: 1080 });
    await screenshotPage.goto(`file:///${path.resolve(screenshotPath)}`);
    
    const diff = await baselinePage.evaluate((screenshotPage) => {
      const baseline = document.querySelector('img');
      const screenshot = screenshotPage.querySelector('img');
      return { width: baseline?.width || 0, height: baseline?.height || 0 };
    }, screenshotPage);
    
    results.push({
      file: baselineFile,
      status: 'COMPARED',
      diff,
    });
    
    await baselinePage.close();
    await screenshotPage.close();
  }
  
  await browser.close();
  
  const reportPath = `preview/results/reports/diff-report-${Date.now()}.json`;
  fs.writeFileSync(reportPath, JSON.stringify(results, null, 2));
  console.log(`Diff report saved to: ${reportPath}`);
}

async function createBaselines() {
  const fs = require('fs');
  const baselineDir = 'preview/results/baselines';
  
  if (!fs.existsSync(baselineDir)) {
    fs.mkdirSync(baselineDir, { recursive: true });
  }
  
  console.log('Creating baseline screenshots...');
  
  const { pcDevices, h5Devices } = require('./config/devices');
  const { pcRoutes, h5Routes } = require('./config/routes');
  const { chromium } = require('playwright');
  
  const browser = await chromium.launch({ headless: true });
  
  const platforms = [
    { name: 'pc', devices: pcDevices, routes: pcRoutes, baseUrl: 'http://localhost:5173' },
    { name: 'h5', devices: h5Devices, routes: h5Routes, baseUrl: 'http://localhost:5174' },
  ];
  
  for (const platform of platforms) {
    for (const device of platform.devices) {
      const context = await browser.newContext({ viewport: device.viewport });
      const page = await context.newPage();
      
      for (const route of platform.routes) {
        const url = platform.name === 'h5' ? `${platform.baseUrl}/#${route.path}` : `${platform.baseUrl}${route.path}`;
        
        try {
          await page.goto(url, { waitUntil: 'networkidle' });
          const filePath = `${baselineDir}/${platform.name}-${device.name.replace(/\s/g, '-')}-${route.name}.png`;
          await page.screenshot({ path: filePath, fullPage: true });
          console.log(`Created baseline: ${filePath}`);
        } catch (error) {
          console.error(`Failed to create baseline for ${platform.name} ${device.name} ${route.name}: ${error.message}`);
        }
      }
      
      await context.close();
    }
  }
  
  await browser.close();
}

const args = process.argv.slice(2);
if (args.includes('--create')) {
  createBaselines().catch(console.error);
} else {
  compareBaselines().catch(console.error);
}
```
